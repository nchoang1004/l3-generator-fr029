    2| 
    3| В tenacity 9.x встроенного CB больше нет, поэтому контур — своя небольшая

    7| import time
    8| from dataclasses import dataclass, field
    9| from enum import StrEnum
   10| 
   11| 
   12| class CircuitState(StrEnum):
   13|     closed = "closed"
   14|     open = "open"
   15|     half_open = "half_open"
   16| 
   17| 
   18| class CircuitOpenError(Exception):
   19|     """Контур разомкнут: вызов делать нельзя, нужен фолбэк."""
   20| 
   21| 
   22| @dataclass
   23| class CircuitBreaker:
   24|     failure_threshold: int = 3
   25|     recovery_timeout: float = 30.0
   26|     _state: CircuitState = field(default=CircuitState.closed, init=False, repr=False)
   27|     _failures: int = field(default=0, init=False, repr=False)
   28|     _opened_at: float | None = field(default=None, init=False, repr=False)
   29| 
   30|     @property
   31|     def state(self) -> CircuitState:
   32|         """Текущее состояние; открытый контур уходит в half_open после паузы."""
   33|         if self._state is CircuitState.open:
   34|             opened_at = self._opened_at
   35|             if opened_at is not None and time.monotonic() - opened_at >= self.recovery_timeout:
   36|                 self._state = CircuitState.half_open
   37|         return self._state
   38| 
   39|     @property
   40|     def failures(self) -> int:
   41|         return self._failures
   42| 
   43|     def before_call(self) -> None:
   44|         if self.state is CircuitState.open:
   45|             raise CircuitOpenError("catalog circuit breaker is open")
   46| 
   47|     def record_success(self) -> None:
   48|         self._state = CircuitState.closed
   49|         self._failures = 0
   50|         self._opened_at = None
   51| 
   52|     def record_failure(self) -> None:
   53|         self._failures += 1
   54|         if self.state is CircuitState.half_open or self._failures >= self.failure_threshold:
   55|             self._state = CircuitState.open

   18| class EventBus(Protocol):
   19|     """Минимальный контракт шины: публикация и подписка на топик."""
   20| 
   21|     async def publish(self, topic: str, payload: Mapping[str, Any]) -> None: ...
   22| 
   23|     async def subscribe(self, topic: str, handler: EventHandler) -> None: ...
   24| 
   25|     async def close(self) -> None: ...
   26| 
   27| 
   28| class InMemoryEventBus:
   29|     """Бус внутри одного процесса: подписчики вызываются синхронно в publish.
   30| 
   31|     Использается в тестах и при локальном запуске без Redis — публикация
   32|     детерминирована: когда POST /orders вернулся, уведомление уже создано.
   33|     """
   34| 
   35|     def __init__(self) -> None:
   36|         self._subscribers: dict[str, list[EventHandler]] = {}
   37|         self.published: list[tuple[str, Mapping[str, Any]]] = []
   38| 
   39|     async def publish(self, topic: str, payload: Mapping[str, Any]) -> None:
   40|         self.published.append((topic, payload))
   41|         for handler in self._subscribers.get(topic, []):
   42|             await handler(payload)
   43| 
   44|     async def subscribe(self, topic: str, handler: EventHandler) -> None:
   45|         self._subscribers.setdefault(topic, []).append(handler)
   46| 
   47|     async def close(self) -> None:
   48|         self._subscribers.clear()

   50| 
   51| class RedisEventBus:
   52|     """Prod-реализация на Redis pub/sub: один канал на топик."""
   53| 
   54|     def __init__(self, client: Redis, *, channel_prefix: str = "orderflow") -> None:
   55|         self._client = client
   56|         self._channel_prefix = channel_prefix
   57|         self._tasks: list[asyncio.Task[None]] = []
   58|         self._pubsubs: list[PubSub] = []
   59| 
   60|     def _channel(self, topic: str) -> str:
   61|         return f"{self._channel_prefix}:{topic}"
   62| 
   63|     async def publish(self, topic: str, payload: Mapping[str, Any]) -> None:
   64|         message = json.dumps(payload, default=str)
   65|         await self._client.publish(self._channel(topic), message)
   66| 
   67|     async def subscribe(self, topic: str, handler: EventHandler) -> None:
   68|         pubsub = self._client.pubsub()
   69|         await pubsub.subscribe(self._channel(topic))
   70|         self._pubsubs.append(pubsub)
   71|         self._tasks.append(asyncio.create_task(self._consume(pubsub, handler)))
   72| 
   73|     async def _consume(self, pubsub: PubSub, handler: EventHandler) -> None:
   74|         try:
   75|             async for message in pubsub.listen():
   76|                 if message.get("type") != "message":
   77|                     continue
   78|                 data = message.get("data")
   79|                 text = data.decode() if isinstance(data, bytes | bytearray) else str(data)
   80|                 await handler(json.loads(text))
   81|         except asyncio.CancelledError:
   82|             raise
   83|         except Exception:
   84|             logger.exception("event consumer failed")
   85| 
   86|     async def close(self) -> None:
   87|         for task in self._tasks:
   88|             task.cancel()
   89|         for task in self._tasks:
   90|             with contextlib.suppress(asyncio.CancelledError):
   91|                 await task
   92|         self._tasks.clear()
   93|         for pubsub in self._pubsubs:
   94|             await pubsub.aclose()  # type: ignore[no-untyped-call]  # redis-py без аннотаций
   95|         self._pubsubs.clear()
   96|         await self._client.aclose()
